## Application description


It is a webpage serving application, where it serves a very big (~53kB) .svg file 
at the /svg endpoint. It is clearly bigger then the TX buffer (16kB).

Give extra attention to DMA handling. The w5500 should use DMA. 


## Problem Summary

First 172.16.1.1/svg is successful, I receive the whole .svg file nothing truncated.
The second call (loading again) fails.

The MCU hard locks with no timeout messages or error output.

The below logs reflects the following:
http://172.16.1.1 loading, which also loads the 172.16.1.1/svg url, which freezes completely.

# Architecture

| Component | Details |
|-----------|---------|
| MCU | RP2350A (Raspberry Pi Pico 2) |
| Ethernet | W5500 via SPI0 @ 20MHz |
| Camera | OV5640 via PIO + DMA |
| Frame size | 240x320 RGB565 = 153.6KB |
| TX buffer | 16KB (Socket 0) |
| RX buffer | 16KB (Socket 0) |

            angle = petal * 45 + layer * 10  # Slight rotation per layer
            scale = 1 - layer * 0.1  # Smaller petals in outer layers

    for thorn in range(10):
        y_pos = 180 + thorn * 10

    uint8_t socket_num;           // Target socket
    tx_state_t state;             // Current state
    struct w5500_tx_transfer* next; // For linked list queue
} w5500_tx_transfer_t;

static void wiz_spi_writeburst(const uint8_t* pBuf, uint16_t len) {
    if (len >= 1024 && dma_tx_chan >= 0) {
        // printf("DMA3 used for %d bytes\n", len);
        machine_spi_obj_t *spi_obj = (machine_spi_obj_t *)wiznet5k_obj.spi;
        spi_inst_t *spi_inst = spi_obj->spi_inst;

        // Claim RX channel to drain SPI RX FIFO
        int dma_rx_chan = dma_claim_unused_channel(true);
        if (dma_rx_chan < 0) {
            // Fallback to normal SPI if no RX channel available
            wiznet5k_obj.spi_transfer(wiznet5k_obj.spi, len, pBuf, NULL);
            return;
        }

        uint8_t dev_null;  // Dummy variable for RX data

        // Configure TX channel
        dma_channel_config c = dma_channel_get_default_config(dma_tx_chan);
        channel_config_set_transfer_data_size(&c, DMA_SIZE_8);
        channel_config_set_read_increment(&c, true);
        channel_config_set_write_increment(&c, false);
        channel_config_set_dreq(&c, spi_get_dreq(spi_inst, true));
        dma_channel_configure(dma_tx_chan, &c, &spi_get_hw(spi_inst)->dr, pBuf, len, false);

        // Configure RX channel to drain RX FIFO
        c = dma_channel_get_default_config(dma_rx_chan);
        channel_config_set_transfer_data_size(&c, DMA_SIZE_8);
        channel_config_set_read_increment(&c, false);
        channel_config_set_write_increment(&c, false);
        channel_config_set_dreq(&c, spi_get_dreq(spi_inst, false));
        dma_channel_configure(dma_rx_chan, &c, &dev_null, &spi_get_hw(spi_inst)->dr, len, false);

        // Enable SPI DMA for transfer
        spi_get_hw(spi_obj->spi_inst)->dmacr = (1 << 1) | (1 << 0);
        __compiler_memory_barrier();
        dma_start_channel_mask((1u << dma_rx_chan) | (1u << dma_tx_chan));
        dma_channel_wait_for_finish_blocking(dma_rx_chan);
        dma_channel_wait_for_finish_blocking(dma_tx_chan);
        __compiler_memory_barrier();
        // Disable SPI DMA
        spi_get_hw(spi_obj->spi_inst)->dmacr = 0;

        dma_channel_unclaim(dma_rx_chan);
        // printf("DMA is finished\n", len);
    } else {
        wiznet5k_obj.spi_transfer(wiznet5k_obj.spi, len, pBuf, NULL);
    }
}

// Initialize transfer pool only (DMA channel claimed earlier)
static void w5500_tx_pool_init(void) {
    // Initialize transfer pool
    for (int i = 0; i < 8; i++) {
        transfer_pool_data[i].next = transfer_pool;
        transfer_pool = &transfer_pool_data[i];
    }
    DEBUG_PRINT("w5500_tx_pool_init: initialized transfer pool\n");
}

// Claim DMA channel early, before any SPI operations
static void w5500_dma_init(void) {
    if (dma_tx_chan < 0) {
        dma_tx_chan = dma_claim_unused_channel(true);
        DEBUG_PRINT("w5500_dma_init: claimed TX channel %d\n", dma_tx_chan);
    }
}

// Get free transfer descriptor from pool
static w5500_tx_transfer_t* w5500_get_transfer(void) {
    if (transfer_pool) {
        w5500_tx_transfer_t* transfer = transfer_pool;
        transfer_pool = transfer->next;
        memset(transfer, 0, sizeof(w5500_tx_transfer_t));
        return transfer;
    }
    return NULL; // Pool exhausted
}

// Return transfer descriptor to pool
static void w5500_release_transfer(w5500_tx_transfer_t* transfer) {
    transfer->next = transfer_pool;
    transfer_pool = transfer;
}

// Add transfer to active queue
static void w5500_queue_transfer(w5500_tx_transfer_t* transfer) {
    transfer->next = active_transfers;
    active_transfers = transfer;
}

// Remove transfer from active queue
static void w5500_dequeue_transfer(w5500_tx_transfer_t* transfer) {
    w5500_tx_transfer_t** current = &active_transfers;
    while (*current) {
        if (*current == transfer) {
            *current = transfer->next;
            return;
        }
        current = &(*current)->next;
    }
}

/*
 * Complete rewrite of w5500_dma_write_to_txbuf
 *
 * This function properly follows the W5500 SPI protocol:
 * 1. Assert CS
 * 2. Send 3-byte header (address high, address low, control byte)
 * 3. Send data payload via DMA
 * 4. Wait for DMA completion
 * 5. Disable SPI DMA mode
 * 6. Deassert CS
 *
 * Parameters:
 *   sn - socket number (0-7)
 *   offset - offset within the TX buffer (handles wraparound internally)
 *   data - pointer to data to write
 *   len - number of bytes to write
 */
static void w5500_dma_write_to_txbuf(uint8_t sn, uint16_t offset, const uint8_t* data, uint16_t len) {
    if (len == 0 || dma_tx_chan < 0) return;

    // Get SPI object
    machine_spi_obj_t *spi_obj = (machine_spi_obj_t *)wiznet5k_obj.spi;
    spi_inst_t *spi_inst = spi_obj->spi_inst;

    // Calculate W5500 address for TX buffer write
    // W5500 address format: [15:0] = offset, sent as: addr_high, addr_low, control
    // Control byte for socket n TX buffer write: (WIZCHIP_TXBUF_BLOCK(sn) << 3) | 0x04
    // The 0x04 indicates write mode (VDM mode)
    uint8_t control = (WIZCHIP_TXBUF_BLOCK(sn) << 3) | 0x04;

    // Build header: address high, address low, control byte
    uint8_t header[3] = {
        (offset >> 8) & 0xFF,   // Address high byte
        offset & 0xFF,          // Address low byte
        control                  // Control byte (block select + write)
    };

    // Claim RX channel to drain SPI RX FIFO during DMA
    int dma_rx_chan = dma_claim_unused_channel(false);  // false = don't panic if unavailable

    // === ASSERT CS ===
    wiz_cs_select();

    // === SEND 3-BYTE HEADER (using normal SPI transfer) ===
    wiznet5k_obj.spi_transfer(wiznet5k_obj.spi, 3, header, NULL);

    // === SEND DATA PAYLOAD VIA DMA ===
    if (dma_rx_chan >= 0 && len >= 64) {
        // Use DMA for larger transfers
        static uint8_t dev_null;  // Dummy variable for RX data

        // Configure TX channel
        dma_channel_config c = dma_channel_get_default_config(dma_tx_chan);
        channel_config_set_transfer_data_size(&c, DMA_SIZE_8);
        channel_config_set_read_increment(&c, true);      // Read from source memory
        channel_config_set_write_increment(&c, false);    // Write to SPI register
        channel_config_set_dreq(&c, spi_get_dreq(spi_inst, true));
        dma_channel_configure(dma_tx_chan, &c, &spi_get_hw(spi_inst)->dr, data, len, false);

        // Configure RX channel to drain RX FIFO
        c = dma_channel_get_default_config(dma_rx_chan);
        channel_config_set_transfer_data_size(&c, DMA_SIZE_8);
        channel_config_set_read_increment(&c, false);
        channel_config_set_write_increment(&c, false);
        channel_config_set_dreq(&c, spi_get_dreq(spi_inst, false));
        dma_channel_configure(dma_rx_chan, &c, &dev_null, &spi_get_hw(spi_inst)->dr, len, false);

        // Enable SPI DMA for transfer
        spi_get_hw(spi_inst)->dmacr = (1 << 1) | (1 << 0);
        __compiler_memory_barrier();

        // Start both DMA channels
        dma_start_channel_mask((1u << dma_rx_chan) | (1u << dma_tx_chan));

        // Wait for completion
        dma_channel_wait_for_finish_blocking(dma_rx_chan);
        dma_channel_wait_for_finish_blocking(dma_tx_chan);

        __compiler_memory_barrier();

        // === DISABLE SPI DMA MODE (CRITICAL!) ===
        spi_get_hw(spi_inst)->dmacr = 0;

        // Release RX channel
        dma_channel_unclaim(dma_rx_chan);
    } else {
        // Use normal SPI transfer for smaller data or if no RX channel available
        wiznet5k_obj.spi_transfer(wiznet5k_obj.spi, len, data, NULL);

        // Release RX channel if we claimed it
        if (dma_rx_chan >= 0) {
            dma_channel_unclaim(dma_rx_chan);
        }
    }

    // === DEASSERT CS ===
    wiz_cs_deselect();

    DEBUG_PRINT("w5500_dma_write_to_txbuf: wrote %d bytes to socket %d offset %d\n", len, sn, offset);
}

// Handle W5500 TX buffer wrap-around for chunk transfers
// Use the corrected w5500_dma_write_to_txbuf function
static void w5500_send_chunk(uint8_t sn, const uint8_t* data, uint16_t len, uint16_t tx_wr_ptr) {
    uint32_t buffer_size = getSn_TxMAX(sn);
    uint16_t buffer_mask = buffer_size - 1;  // Assumes power of 2
    uint16_t offset = tx_wr_ptr & buffer_mask;
    uint16_t space_to_end = buffer_size - offset;

    DEBUG_PRINT("w5500_send_chunk: sn=%d, len=%d, tx_wr_ptr=%d, offset=%d, space_to_end=%d\n",
                sn, len, tx_wr_ptr, offset, space_to_end);

    if (len <= space_to_end) {
        // Single transfer - no wraparound needed
        w5500_dma_write_to_txbuf(sn, offset, data, len);
    } else {
        // Split across buffer boundary (wraparound)
        // First part: write to end of buffer
        w5500_dma_write_to_txbuf(sn, offset, data, space_to_end);

        // Second part: write from start of buffer
        w5500_dma_write_to_txbuf(sn, 0, data + space_to_end, len - space_to_end);
    }
}

/*
 * Drain any remaining data from socket RX buffer.
 * This must be called before re-listening to ensure clean state.
 *
 * Parameters:
 *   sn - socket number (0-7)
 *
 * Returns:
 *   Total bytes drained from buffer
 */
static uint32_t w5500_drain_rx_buffer(uint8_t sn) {
    uint32_t total_drained = 0;
    uint16_t rx_size;

    // Keep reading until RX buffer is empty
    while ((rx_size = getSn_RX_RSR(sn)) > 0) {
        // Calculate how much to read this iteration (max 256 bytes at a time)
        uint16_t to_read = (rx_size > 256) ? 256 : rx_size;

        // We don't need the data, just need to advance the read pointer
        // Read into a small static buffer and discard
        static uint8_t drain_buf[256];

        // Use WIZCHIP_EXPORT(recv) to properly update RX pointers
        mp_int_t ret = WIZCHIP_EXPORT(recv)(sn, drain_buf, to_read);

        if (ret <= 0) {
            // Error or no more data - stop draining
            DEBUG_PRINT("w5500_drain_rx_buffer: recv returned %d, stopping\\n", ret);
            break;
        }

        total_drained += ret;

        // Safety limit - don't drain more than 64KB (prevents infinite loop)
        if (total_drained > 65536) {
            DEBUG_PRINT("w5500_drain_rx_buffer: safety limit reached\\n");
            break;
        }
    }

    if (total_drained > 0) {
        DEBUG_PRINT("w5500_drain_rx_buffer: drained %d bytes from socket %d\\n",
                    total_drained, sn);
    }

    return total_drained;
}

// Background transfer service - called periodically
static void w5500_tx_service(void) {
    // Prevent reentrant calls that cause W5500 double-SEND corruption
    if (tx_service_active) {
        return;
    }
    tx_service_active = true;

    w5500_tx_transfer_t* transfer = active_transfers;

    while (transfer) {
        w5500_tx_transfer_t* next = transfer->next; // Save next pointer in case transfer is removed

        // NEW: Validate socket is still in a valid state for transfers
        uint8_t sr = getSn_SR(transfer->socket_num);
        if (sr != SOCK_ESTABLISHED && sr != SOCK_CLOSE_WAIT) {
            DEBUG_PRINT("w5500_tx_service: socket %d no longer valid (state=0x%02x), aborting transfer\n",
                        transfer->socket_num, sr);
            transfer->state = TX_ERROR;
            w5500_dequeue_transfer(transfer);
            w5500_release_transfer(transfer);
            transfer = next;
            continue;
        }
        
        switch (transfer->state) {
            case TX_WAITING_BUFFER_SPACE: {
                uint16_t free_space = getSn_TX_FSR(transfer->socket_num);
                uint8_t sr = getSn_SR(transfer->socket_num);

                if ((sr != SOCK_ESTABLISHED) && (sr != SOCK_CLOSE_WAIT)) {
                    DEBUG_PRINT("w5500_tx_service: socket %d not connected, cancelling transfer\n", transfer->socket_num);
                    transfer->state = TX_ERROR;
                    break;
                }

                if (free_space >= transfer->chunk_size) {
                    // Start DMA transfer
                    uint16_t tx_wr = getSn_TX_WR(transfer->socket_num);
                    const uint8_t* data_ptr = transfer->data_ptr + transfer->bytes_sent;

                    w5500_send_chunk(transfer->socket_num, data_ptr, transfer->chunk_size, tx_wr);
                    transfer->state = TX_TRANSFERRING_CHUNK;

                    DEBUG_PRINT("w5500_tx_service: started DMA for %d bytes on socket %d\n",
                                transfer->chunk_size, transfer->socket_num);
                }
                break;
            }

            case TX_TRANSFERRING_CHUNK: {
                // Transfer is now synchronous, so we can immediately proceed
                // DMA is completed before w5500_send_chunk returns

                // Update write pointer and send command
                uint16_t new_tx_wr = getSn_TX_WR(transfer->socket_num) + transfer->chunk_size;
                setSn_TX_WR(transfer->socket_num, new_tx_wr);
                setSn_CR(transfer->socket_num, Sn_CR_SEND);

                // CRITICAL: Change state BEFORE yield to prevent double-SEND on reentry
                transfer->state = TX_SENDING_PACKET;

                // Wait for command to register
                while (getSn_CR(transfer->socket_num)) {
                    mpy_wiznet_yield();
                }
                DEBUG_PRINT("w5500_tx_service: DMA complete, SEND command issued on socket %d\n",
                            transfer->socket_num);
                break;
            }

            case TX_SENDING_PACKET: {
                uint8_t ir = getSn_IR(transfer->socket_num);
                if (ir & Sn_IR_SENDOK) {
                    setSn_IR(transfer->socket_num, Sn_IR_SENDOK);
                    
                    transfer->bytes_sent += transfer->chunk_size;
                    DEBUG_PRINT("w5500_tx_service: SENDOK, %d/%d bytes sent on socket %d\n", 
                                transfer->bytes_sent, transfer->total_size, transfer->socket_num);
                    
                    if (transfer->bytes_sent >= transfer->total_size) {
                        transfer->state = TX_COMPLETE;
                    } else {
                        // Calculate next chunk size
                        uint16_t remaining = transfer->total_size - transfer->bytes_sent;
                        uint16_t buffer_free = getSn_TX_FSR(transfer->socket_num);
                        transfer->chunk_size = (remaining < buffer_free) ? remaining : buffer_free;
                        // Cap chunk size to buffer size
                        uint16_t max_chunk = getSn_TxMAX(transfer->socket_num);
                        if (transfer->chunk_size > max_chunk) {
                            transfer->chunk_size = max_chunk;
                        }
                        transfer->state = TX_WAITING_BUFFER_SPACE;
                    }
                } else if (ir & Sn_IR_TIMEOUT) {
                    DEBUG_PRINT("w5500_tx_service: TIMEOUT on socket %d\n", transfer->socket_num);
                    transfer->state = TX_ERROR;
                }
                break;
            }
            
            case TX_COMPLETE:
                DEBUG_PRINT("w5500_tx_service: Transfer complete on socket %d, %d bytes sent\n", 
                            transfer->socket_num, transfer->bytes_sent);
                w5500_dequeue_transfer(transfer);
                w5500_release_transfer(transfer);
                break;
                
            case TX_ERROR:
                DEBUG_PRINT("w5500_tx_service: Transfer error on socket %d\n", transfer->socket_num);
                w5500_dequeue_transfer(transfer);
                w5500_release_transfer(transfer);
                break;
                
            default:
                break;
        }

        transfer = next;
    }

    tx_service_active = false;
}

// CLAUDE FIX: Helper function to wait for a specific transfer to complete
static int w5500_wait_transfer_complete(w5500_tx_transfer_t* transfer, uint32_t timeout_ms) {
    uint32_t start = mp_hal_ticks_ms();

    while (transfer->state != TX_COMPLETE && transfer->state != TX_ERROR) {
        w5500_tx_service();

        if (timeout_ms > 0 && (mp_hal_ticks_ms() - start) >= timeout_ms) {
            DEBUG_PRINT("w5500_wait_transfer_complete: timeout\n");
            return -1;  // Timeout
        }

        mp_hal_delay_ms(1);
    }

    return (transfer->state == TX_COMPLETE) ? 0 : -1;
}

static void wiznet5k_send_ethernet(wiznet5k_obj_t *self, size_t len, const uint8_t *buf) {
    uint8_t ip[4] = {1, 1, 1, 1}; // dummy
    int ret = WIZCHIP_EXPORT(sendto)(0, (byte *)buf, len, ip, 11); // dummy port
    if (ret != len) {
        printf("wiznet5k_send_ethernet: fatal error %d\n", ret);
        netif_set_link_down(&self->netif);
        netif_set_down(&self->netif);
    }
}

// Stores the frame in self->eth_frame and returns number of bytes in the frame, 0 for no frame
static uint16_t wiznet5k_recv_ethernet(wiznet5k_obj_t *self) {
    uint16_t len = getSn_RX_RSR(0);
    if (len == 0) {
        return 0;
    }

    byte ip[4];
    uint16_t port;
    int ret = WIZCHIP_EXPORT(recvfrom)(0, self->eth_frame, 1514, ip, &port);
    if (ret <= 0) {
        printf("wiznet5k_recv_ethernet: fatal error len=%u ret=%d\n", len, ret);
        netif_set_link_down(&self->netif);
        netif_set_down(&self->netif);
        return 0;
    }

    return ret;
}

static err_t wiznet5k_netif_output(struct netif *netif, struct pbuf *p) {
    wiznet5k_obj_t *self = netif->state;
    const uint8_t *buf = self->eth_frame;
    if (p->next == NULL && p->len == p->tot_len) {
        buf = p->payload;
    } else {
        pbuf_copy_partial(p, self->eth_frame, p->tot_len, 0);
    }
    if (self->trace_flags & TRACE_ETH_TX) {
        netutils_ethernet_trace(MP_PYTHON_PRINTER, p->tot_len, buf, NETUTILS_TRACE_IS_TX | NETUTILS_TRACE_NEWLINE);
    }
    wiznet5k_send_ethernet(self, p->tot_len, buf);
    return ERR_OK;
}

static err_t wiznet5k_netif_init(struct netif *netif) {
    netif->linkoutput = wiznet5k_netif_output;
    netif->output = etharp_output;
    netif->mtu = 1500;
    netif->flags = NETIF_FLAG_BROADCAST | NETIF_FLAG_ETHARP | NETIF_FLAG_ETHERNET | NETIF_FLAG_IGMP;
    wiznet5k_get_mac_address(netif->state, netif->hwaddr);
    netif->hwaddr_len = sizeof(netif->hwaddr);
    int ret = WIZCHIP_EXPORT(socket)(0, Sn_MR_MACRAW, 0, 0);
    if (ret != 0) {
        printf("WIZNET fatal error in netif_init: %d\n", ret);
        return ERR_IF;
    }

    // Enable MAC filtering so we only get frames destined for us, to reduce load on lwIP
    setSn_MR(0, getSn_MR(0) | Sn_MR_MFEN);

    #if LWIP_IPV6
    netif->output_ip6 = ethip6_output;
    netif->flags |= NETIF_FLAG_MLD6;
    #else
    // Drop IPv6 packets if firmware does not support it
    setSn_MR(0, getSn_MR(0) | Sn_MR_MIP6B);
    #endif

    return ERR_OK;
}

static void wiznet5k_lwip_init(wiznet5k_obj_t *self) {
    ip_addr_t ipconfig[4];
    IP_ADDR4(&ipconfig[0], 0, 0, 0, 0);
    IP_ADDR4(&ipconfig[1], 0, 0, 0, 0);
    IP_ADDR4(&ipconfig[2], 0, 0, 0, 0);
    IP_ADDR4(&ipconfig[3], 0, 0, 0, 0);
    netif_add(&self->netif, ip_2_ip4(&ipconfig[0]), ip_2_ip4(&ipconfig[1]), ip_2_ip4(&ipconfig[2]), self, wiznet5k_netif_init, ethernet_input);
    self->netif.name[0] = 'e';
    self->netif.name[1] = '0';
    netif_set_default(&self->netif);
    dns_setserver(0, &ipconfig[3]);
    dhcp_set_struct(&self->netif, &self->dhcp_struct);
    // Setting NETIF_FLAG_UP then clearing it is a workaround for dhcp_start and the
    // LWIP_DHCP_CHECK_LINK_UP option, so that the DHCP client schedules itself to
    // automatically start when the interface later goes up.
    self->netif.flags |= NETIF_FLAG_UP;
    dhcp_start(&self->netif);
    self->netif.flags &= ~NETIF_FLAG_UP;

    #if LWIP_MDNS_RESPONDER
    // NOTE: interface is removed in ::wiznet5k_deinit(), which is called as
    // part of the init sequence.
    mdns_resp_add_netif(&self->netif, mod_network_hostname_data);
    #endif
}

void wiznet5k_poll(void) {
    wiznet5k_obj_t *self = &wiznet5k_obj;
    if ((self->netif.flags & (NETIF_FLAG_UP | NETIF_FLAG_LINK_UP)) == (NETIF_FLAG_UP | NETIF_FLAG_LINK_UP)) {
        uint16_t len;
        while ((len = wiznet5k_recv_ethernet(self)) > 0) {
            if (self->trace_flags & TRACE_ETH_RX) {
                netutils_ethernet_trace(MP_PYTHON_PRINTER, len, self->eth_frame, NETUTILS_TRACE_NEWLINE);
            }
            struct pbuf *p = pbuf_alloc(PBUF_RAW, len, PBUF_POOL);
            if (p != NULL) {
                pbuf_take(p, self->eth_frame, len);
                if (self->netif.input(p, &self->netif) != ERR_OK) {
                    pbuf_free(p);
                }
            }
        }
    }
    wizchip_clrinterrupt(IK_SOCK_0);
    #if _WIZCHIP_ == W5100S
    setSn_IR(0, Sn_IR_RECV); // W5100S driver bug: must write to the Sn_IR register to reset the IRQ signal
    #endif
}

static void wiznet5k_init(void) {
    DEBUG_PRINT("wiznet5k_init: initializing with provided TCP stack\n");

    // CLAUDE FIX: Claim DMA channel FIRST, before any SPI operations
    w5500_dma_init();

    // Configure wiznet provided TCP / socket interface

    reg_dhcp_cbfunc(wiz_dhcp_assign, wiz_dhcp_update, wiz_dhcp_conflict);

    // 16k RX + 16k TX, the correct order is:
    // [ RX0, RX1, RX2, RX3, RX4, RX5, RX6, RX7,
    // TX0, TX1, TX2, TX3, TX4, TX5, TX6, TX7 ]
    uint8_t sn_size[16] = {16, 0, 0, 0, 0, 0, 0, 0, 16, 0, 0, 0, 0, 0, 0, 0};
    DEBUG_PRINT("wiznet5k_init: calling ctlwizchip(CW_INIT_WIZCHIP, ...)\n");
    ctlwizchip(CW_INIT_WIZCHIP, sn_size);

    DEBUG_PRINT("wiznet5k_init: calling ctlnetwork(CN_SET_NETINFO, ...)\n");
    ctlnetwork(CN_SET_NETINFO, (void *)&wiznet5k_obj.netinfo);

    // set some sensible default values; they are configurable using ifconfig method
    wiz_NetInfo netinfo = {
        .mac = {0, 0, 0, 0, 0, 0},
        .ip = {192, 168, 0, 18},
        .sn = {255, 255, 255, 0},
        .gw = {192, 168, 0, 1},
        .dns = {8, 8, 8, 8}, // Google public DNS
        .dhcp = NETINFO_STATIC,
    };
    wiznet5k_obj.netinfo = netinfo;

    DEBUG_PRINT("wiznet5k_init: registering with network module\n");
    // register with network module
    mod_network_register_nic(&wiznet5k_obj);

    // Initialize socket state tracking
    for (int i = 0; i < _WIZCHIP_SOCK_NUM_; i++) {
        wiznet5k_obj.socket_states[i].port = 0;
        wiznet5k_obj.socket_states[i].type = 0;
        wiznet5k_obj.socket_states[i].is_listening = false;
    }

    wiznet5k_obj.active = true;

    // Initialize transfer pool (DMA channel already claimed above)
    w5500_tx_pool_init();

    DEBUG_PRINT("wiznet5k_init: done, active=true\n");
}

static int wiznet5k_gethostbyname(mp_obj_t nic, const char *name, mp_uint_t len, uint8_t *out_ip) {
    uint8_t dns_ip[MOD_NETWORK_IPADDR_BUF_SIZE] = {8, 8, 8, 8};
    uint8_t *buf = m_new(uint8_t, MAX_DNS_BUF_SIZE);
    DNS_init(2, buf);
    if (wiznet5k_obj.netinfo.dns[0]) {
        memcpy(dns_ip, wiznet5k_obj.netinfo.dns, MOD_NETWORK_IPADDR_BUF_SIZE);
    }
    mp_int_t ret = DNS_run(dns_ip, (uint8_t *)name, out_ip);
    m_del(uint8_t, buf, MAX_DNS_BUF_SIZE);
    if (ret == 1) {
        // success
        return 0;
    } else {
        // failure
        return -2;
    }
}

static int wiznet5k_socket_socket(mod_network_socket_obj_t *socket, int *_errno) {
    DEBUG_PRINT("socket_socket: domain=%d, type=%d, proto=%d\n",
                socket->domain, socket->type, socket->proto);

    if (socket->domain != MOD_NETWORK_AF_INET) {
        DEBUG_PRINT("socket_socket: unsupported domain %d\n", socket->domain);
        *_errno = MP_EAFNOSUPPORT;
        return -1;
    }

    switch (socket->type) {
        case MOD_NETWORK_SOCK_STREAM:
            socket->type = Sn_MR_TCP;
            DEBUG_PRINT("socket_socket: TCP socket\n");
            break;
        case MOD_NETWORK_SOCK_DGRAM:
            socket->type = Sn_MR_UDP;
            DEBUG_PRINT("socket_socket: UDP socket\n");
            break;
        default:
            DEBUG_PRINT("socket_socket: unsupported type %d\n", socket->type);
            *_errno = MP_EINVAL;
            return -1;
    }

    if (socket->fileno == -1) {
        // get first unused socket number
        for (mp_uint_t sn = 0; sn < _WIZCHIP_SOCK_NUM_; sn++) {
            if ((wiznet5k_obj.socket_used & (1 << sn)) == 0) {
                wiznet5k_obj.socket_used |= (1 << sn);
                socket->fileno = sn;
                DEBUG_PRINT("socket_socket: assigned socket number %d\n", sn);
                break;
            }
        }
        if (socket->fileno == -1) {
            DEBUG_PRINT("socket_socket: no free sockets (socket_used=0x%02x)\n",
                        wiznet5k_obj.socket_used);
            *_errno = MP_EMFILE;
            return -1;
        }
    }

    // WIZNET does not have a concept of pure "open socket".  You need to know
    // if it's a server or client at the time of creation of the socket.
    // So, we defer the open until we know what kind of socket we want.

    // use "domain" to indicate that this socket has not yet been opened
    socket->domain = 0;
    DEBUG_PRINT("socket_socket: success, fileno=%d\n", socket->fileno);

    return 0;
}


static void wiznet5k_socket_close(mod_network_socket_obj_t *socket) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_close: socket %d\n", sn);

    if (sn < _WIZCHIP_SOCK_NUM_) {
        // NEW: Cancel any active transfers on this socket FIRST
        w5500_tx_transfer_t* transfer = active_transfers;
        w5500_tx_transfer_t* prev = NULL;

        while (transfer) {
            w5500_tx_transfer_t* next = transfer->next;

            if (transfer->socket_num == sn) {
                DEBUG_PRINT("socket_close: cancelling active transfer on socket %d\n", sn);

                // Remove from active list
                if (prev) {
                    prev->next = next;
                } else {
                    active_transfers = next;
                }

                // Mark as error and release
                transfer->state = TX_ERROR;
                w5500_release_transfer(transfer);
            } else {
                prev = transfer;
            }

            transfer = next;
        }


        DEBUG_PRINT("socket_close: calling WIZCHIP_EXPORT(close) on socket %d\n", sn);

        // Check state before closing
        uint8_t sr_before = getSn_SR(sn);
        DEBUG_PRINT("socket_close: before - state=0x%02x (%s)\n",
                   sr_before, socket_state_str(sr_before));

        // Get saved socket state if available
        uint16_t saved_port = wiznet5k_obj.socket_states[sn].port;
        uint8_t saved_type = wiznet5k_obj.socket_states[sn].type;
        bool was_listening = wiznet5k_obj.socket_states[sn].is_listening;
        bool is_established = (sr_before == SOCK_ESTABLISHED);
        bool can_relisten = was_listening && (is_established || sr_before == SOCK_CLOSE_WAIT);

        DEBUG_PRINT("socket_close: saved state - port=%d, type=0x%02x, was_listening=%d\n",
                   saved_port, saved_type, was_listening);

        wiznet5k_obj.socket_used &= ~(1 << sn);
        mp_hal_delay_ms(50);
        // CHANGE: Try DISCON first for graceful shutdown if ESTABLISHED
        if (getSn_SR(sn) == SOCK_ESTABLISHED) {
             WIZCHIP_EXPORT(disconnect)(sn);
             // Wait for transition to CLOSED or TIME_WAIT, with timeout
             uint32_t start = mp_hal_ticks_ms();
             while (getSn_SR(sn) != SOCK_CLOSED && mp_hal_ticks_ms() - start < 250) {
                 mp_hal_delay_ms(5);
             }
        }
        WIZCHIP_EXPORT(close)(sn);
        // Give it a moment to transition to CLOSED state
        mp_hal_delay_ms(50);

        // Verify socket reached CLOSED state
        uint8_t sr_after_close = getSn_SR(sn);
        int close_wait_count = 0;
        while (sr_after_close != SOCK_CLOSED && close_wait_count < 20) {
            mp_hal_delay_ms(25);
            sr_after_close = getSn_SR(sn);
            close_wait_count++;
        }

        if (sr_after_close != SOCK_CLOSED) {
            DEBUG_PRINT("socket_close: WARNING - socket not fully closed, state=0x%02x\n",
                        sr_after_close);
        }

        // Check state after closing
        uint8_t sr_after = getSn_SR(sn);
        DEBUG_PRINT("socket_close: after - state=0x%02x (%s)\n",
                   sr_after, socket_state_str(sr_after));

        // Re-listen if this was a listening socket that accepted a connection
        if (can_relisten && saved_port != 0) {
            DEBUG_PRINT("socket_close: re-listening on port %d\n", saved_port);

            // CRITICAL FIX: Drain any remaining RX data before re-listening
            // This prevents stale data from the previous connection polluting
            // the new connection's RX buffer
            uint8_t sr_before_drain = getSn_SR(sn);
            if (sr_before_drain != SOCK_CLOSED) {
                // Socket not fully closed yet - drain the buffer first
                uint32_t drained = w5500_drain_rx_buffer(sn);
                if (drained > 0) {
                    DEBUG_PRINT("socket_close: drained %d stale bytes before re-listen\n", drained);
                }
            }

            mp_int_t ret = WIZCHIP_EXPORT(socket)(sn, saved_type, saved_port, 0);
            DEBUG_PRINT("socket_close: WIZCHIP_EXPORT(socket) returned %d\n", ret);

            if (ret >= 0) {
                // Verify buffer state is clean after socket reopen
                uint16_t rx_rsr = getSn_RX_RSR(sn);
                uint16_t tx_fsr = getSn_TX_FSR(sn);
                uint16_t tx_max = getSn_TxMAX(sn);

                DEBUG_PRINT("socket_close: after reopen - RX_RSR=%d, TX_FSR=%d/%d\n",
                            rx_rsr, tx_fsr, tx_max);

                // RX should be 0, TX should be at maximum
                if (rx_rsr != 0 || tx_fsr != tx_max) {
                    DEBUG_PRINT("socket_close: WARNING - buffers not clean after reopen!\n");

                    // Force a small delay to let W5500 settle
                    mp_hal_delay_ms(100);

                    // Re-check
                    rx_rsr = getSn_RX_RSR(sn);
                    tx_fsr = getSn_TX_FSR(sn);
                    DEBUG_PRINT("socket_close: after settle - RX_RSR=%d, TX_FSR=%d/%d\n",
                                rx_rsr, tx_fsr, tx_max);
                }
                ret = WIZCHIP_EXPORT(listen)(sn);
                DEBUG_PRINT("socket_close: WIZCHIP_EXPORT(listen) returned %d\n", ret);

                if (ret >= 0) {
                    // Mark socket as in-use and listening again
                    wiznet5k_obj.socket_used |= (1 << sn);
                    wiznet5k_obj.socket_states[sn].is_listening = true;

                    uint8_t sr_final = getSn_SR(sn);
                    if (sr_final == SOCK_LISTEN) {
                        DEBUG_PRINT("socket_close: successfully re-listened\n");
                    } else {
                        DEBUG_PRINT("socket_close: failed to reach LISTEN state, got 0x%02x\n", sr_final);
                        wiznet5k_obj.socket_states[sn].is_listening = false;
                    }
                } else {
                    DEBUG_PRINT("socket_close: listen failed, error=%d\n", ret);
                    wiznet5k_obj.socket_states[sn].is_listening = false;
                }
            } else {
                DEBUG_PRINT("socket_close: socket failed, error=%d\n", ret);
            }
        } else {
            // Clear listening state if we're not re-listening
            if (!is_established) {
                wiznet5k_obj.socket_states[sn].is_listening = false;
                wiznet5k_obj.socket_states[sn].port = 0;
            }
        }
    } else {
        DEBUG_PRINT("socket_close: invalid socket number %d\n", sn);
    }
}

static int wiznet5k_socket_bind(mod_network_socket_obj_t *socket, byte *ip, mp_uint_t port, int *_errno) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_bind: socket=%d, port=%d, type=0x%02x\n",
                sn, port, socket->type);

    // Check if socket is already opened/bound
    uint8_t sr_before = getSn_SR(sn);
    if (sr_before != SOCK_CLOSED && sr_before != 0x00) {
        DEBUG_PRINT("socket_bind: socket already in state 0x%02x (%s)\n",
                    sr_before, socket_state_str(sr_before));
        *_errno = MP_EINVAL;
        return -1;
    }

    // Open the socket in server mode (if port != 0)
    mp_int_t ret = WIZCHIP_EXPORT(socket)(socket->fileno, socket->type, port, 0);
    DEBUG_PRINT("socket_bind: WIZCHIP_EXPORT(socket) returned %d\n", ret);

    if (ret < 0) {
        DEBUG_PRINT("socket_bind: failed, closing socket\n");
        wiznet5k_socket_close(socket);
        *_errno = -ret;
        return -1;
    }

    // Check socket state - should be INIT after successful bind
    uint8_t sr = getSn_SR(sn);
    DEBUG_PRINT("socket_bind: socket %d state after open: 0x%02x (%s)\n",
                sn, sr, socket_state_str(sr));

    if (sr != SOCK_INIT) {
        DEBUG_PRINT("socket_bind: socket not in INIT state after bind\n");
        wiznet5k_socket_close(socket);
        *_errno = MP_EIO;
        return -1;
    }

    // Indicate that this socket has been opened
    socket->domain = 1;
    DEBUG_PRINT("socket_bind: success\n");

    return 0;
}

static int wiznet5k_socket_listen(mod_network_socket_obj_t *socket, mp_int_t backlog, int *_errno) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_listen: socket=%d, backlog=%d\n", sn, backlog);

    // Check current socket state - must be INIT (after bind)
    uint8_t sr_before = getSn_SR(sn);
    DEBUG_PRINT("socket_listen: before - socket %d state: 0x%02x (%s)\n",
                sn, sr_before, socket_state_str(sr_before));

    // Don't allow listen on already connected or closed sockets
    if (sr_before == SOCK_ESTABLISHED || sr_before == SOCK_CLOSE_WAIT) {
        DEBUG_PRINT("socket_listen: socket already connected, cannot listen\n");
        *_errno = MP_EINVAL;
        return -1;
    }

    mp_int_t ret = WIZCHIP_EXPORT(listen)(socket->fileno);
    DEBUG_PRINT("socket_listen: WIZCHIP_EXPORT(listen) returned %d\n", ret);

    if (ret < 0) {
        DEBUG_PRINT("socket_listen: failed, closing socket\n");
        wiznet5k_socket_close(socket);
        *_errno = -ret;
        return -1;
    }

    // Check socket state after listen - should be in LISTEN state
    uint8_t sr_after = getSn_SR(sn);
    DEBUG_PRINT("socket_listen: after - socket %d state: 0x%02x (%s)\n",
                sn, sr_after, socket_state_str(sr_after));

    if (sr_after != SOCK_LISTEN) {
        DEBUG_PRINT("socket_listen: socket not in LISTEN state after listen call\n");
        wiznet5k_socket_close(socket);
        *_errno = MP_EIO;
        return -1;
    }

    // Save socket state for potential re-listening
    wiznet5k_obj.socket_states[sn].port = getSn_PORT(sn);
    wiznet5k_obj.socket_states[sn].type = socket->type & 0x0F;
    wiznet5k_obj.socket_states[sn].is_listening = true;
    DEBUG_PRINT("socket_listen: saved state - port=%d, type=0x%02x\n",
           wiznet5k_obj.socket_states[sn].port, wiznet5k_obj.socket_states[sn].type);

    DEBUG_PRINT("socket_listen: success\n");
    return 0;
}

// Accept a connection on a listening socket.
// NOTE: This implementation is for single-client server mode only.
// The listening socket becomes the connected socket and cannot accept more connections.
// This design maximizes throughput by leveraging the full 16kB RX/TX buffers.
static int wiznet5k_socket_accept(mod_network_socket_obj_t *socket, mod_network_socket_obj_t *socket2, byte *ip, mp_uint_t *port, int *_errno) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_accept: socket=%d, timeout=%d\n", sn, socket->timeout);

    // Check initial socket state - must be in LISTEN or ESTABLISHED state
    // (ESTABLISHED is valid because W5500 may auto-accept connections)
    uint8_t sr_initial = getSn_SR(sn);
    DEBUG_PRINT("socket_accept: initial state - socket %d: 0x%02x (%s)\n",
                sn, sr_initial, socket_state_str(sr_initial));

    if (sr_initial != SOCK_LISTEN && sr_initial != SOCK_ESTABLISHED) {
        DEBUG_PRINT("socket_accept: socket not in LISTEN or ESTABLISHED state (0x%02x)\n", sr_initial);

        // If socket is in CLOSED state but we have a saved port from previous listening,
        // try to re-listen on the same port
        if (sr_initial == SOCK_CLOSED && socket->domain == 1) {
            DEBUG_PRINT("socket_accept: socket in CLOSED state, attempting to re-listen\n");

            uint16_t saved_port = wiznet5k_obj.socket_states[sn].port;
            uint8_t saved_type = wiznet5k_obj.socket_states[sn].type;

            if (saved_port != 0) {
                DEBUG_PRINT("socket_accept: re-listening on port %d\n", saved_port);

                mp_int_t ret = WIZCHIP_EXPORT(socket)(sn, saved_type, saved_port, 0);
                if (ret >= 0) {
                    ret = WIZCHIP_EXPORT(listen)(sn);
                    if (ret >= 0) {
                        uint8_t sr = getSn_SR(sn);
                        if (sr == SOCK_LISTEN) {
                            DEBUG_PRINT("socket_accept: successfully re-listened\n");
                            sr_initial = sr; // Update state and continue
                        } else {
                            DEBUG_PRINT("socket_accept: failed to re-listen, state=0x%02x\n", sr);
                            *_errno = MP_EINVAL;
                            return -1;
                        }
                    } else {
                        DEBUG_PRINT("socket_accept: listen() failed: %d\n", ret);
                        *_errno = -ret;
                        return -1;
                    }
                } else {
                    DEBUG_PRINT("socket_accept: socket() failed: %d\n", ret);
                    *_errno = -ret;
                    return -1;
                }
            }
        } else {
            *_errno = MP_EINVAL;
            return -1;
        }
    }

    // For single-client server: Check if socket is already connected
    // (This shouldn't happen if socket is in LISTEN state, but check anyway)
    uint8_t ir = getSn_IR(sn);
    uint8_t sr = getSn_SR(sn);

    // If socket is already in ESTABLISHED state, accept immediately
    if (sr == SOCK_ESTABLISHED) {
        DEBUG_PRINT("socket_accept: socket already ESTABLISHED, accepting immediately\n");

        // Get client info
        getSn_DIPR(sn, ip);
        *port = getSn_DPORT(sn);

        DEBUG_PRINT("socket_accept: client IP: %d.%d.%d.%d, port: %d\n",
                   ip[0], ip[1], ip[2], ip[3], *port);

        // Single-client server: The listening socket becomes the connected socket
        socket2->domain = socket->domain;
        socket2->type = socket->type;
        socket2->fileno = sn;
        socket2->timeout = socket->timeout;

        DEBUG_PRINT("socket_accept: success (single-client mode)\n");
        return 0;
    }

    // Non-blocking mode: return immediately if no connection pending
    if (socket->timeout == 0) {
        DEBUG_PRINT("socket_accept: non-blocking, no connection pending\n");
        *_errno = MP_EAGAIN;
        return -1;
    }

    uint32_t start_time = mp_hal_ticks_ms();
    uint32_t timeout_ms = socket->timeout;
    if (timeout_ms == -1) {
        timeout_ms = 0x7FFFFFFF; // Use safe max value instead of 0xFFFFFFFF
    }

    int loop_counter = 0;

    for (;;) {
        loop_counter++;

        // Check for timeout (only if not blocking forever)
        if (socket->timeout != -1 && (mp_hal_ticks_ms() - start_time) >= timeout_ms) {
            DEBUG_PRINT("socket_accept: timeout after %d ms\n", timeout_ms);
            *_errno = MP_ETIMEDOUT;
            return -1;
        }

        // Check interrupt register for connection pending
        ir = getSn_IR(sn);
        sr = getSn_SR(sn);

        if (loop_counter % 100 == 0) {
            DEBUG_PRINT("socket_accept: loop %d - socket %d IR: 0x%02x, SR: 0x%02x (%s)\n",
                    loop_counter, sn, ir, sr, socket_state_str(sr));
        }

        // Check if connection is established
        if ((ir & Sn_IR_CON) || sr == SOCK_ESTABLISHED) {
            // Clear the CON interrupt if set
            if (ir & Sn_IR_CON) {
                setSn_IR(sn, Sn_IR_CON);
            }

            // Give it a moment to transition to ESTABLISHED state
            mp_hal_delay_ms(10);

            // Verify we're now in ESTABLISHED state
            sr = getSn_SR(sn);
            if (sr == SOCK_ESTABLISHED) {
                DEBUG_PRINT("socket_accept: socket ESTABLISHED!\n");

                // CRITICAL FIX: Check RX buffer state for debugging
                uint16_t rx_rsr = getSn_RX_RSR(sn);
                uint16_t tx_fsr = getSn_TX_FSR(sn);
                DEBUG_PRINT("socket_accept: RX_RSR=%d, TX_FSR=%d (expected: RX=0, TX=16384)\n",
                            rx_rsr, tx_fsr);

                // If RX buffer already has data, this is suspicious
                // (data arrived before we officially accepted - this is normal)
                // But if TX_FSR is not at max, we have stale TX state
                if (tx_fsr < getSn_TxMAX(sn)) {
                    DEBUG_PRINT("socket_accept: WARNING - TX buffer not fully free!\n");
                }

                // Get client info
                getSn_DIPR(sn, ip);
                *port = getSn_DPORT(sn);

                DEBUG_PRINT("socket_accept: client IP: %d.%d.%d.%d, port: %d\n",
                           ip[0], ip[1], ip[2], ip[3], *port);

                // Single-client server: The listening socket becomes the connected socket
                // Simply copy the socket info to socket2 and mark as connected
                socket2->domain = socket->domain;
                socket2->type = socket->type;
                socket2->fileno = sn;
                socket2->timeout = socket->timeout;

                DEBUG_PRINT("socket_accept: success (single-client mode)\n");
                return 0;
            } else {
                DEBUG_PRINT("socket_accept: socket not in ESTABLISHED state after interrupt\n");
                // Continue waiting
            }
        }

        // Check if socket has closed unexpectedly
        if (sr == SOCK_CLOSED) {
            DEBUG_PRINT("socket_accept: socket closed unexpectedly\n");
            *_errno = MP_ENOTCONN;
            return -1;
        }

        // Only print every 100 loops to reduce noise
        if (loop_counter % 100 == 0) {
            DEBUG_PRINT("socket_accept: still waiting... (loop %d)\n", loop_counter);
        }

        // Wait 50ms to reduce CPU usage
        mp_hal_delay_ms(50);

        // Yield to prevent busy-waiting
        mpy_wiznet_yield();
    }
}

static int wiznet5k_socket_connect(mod_network_socket_obj_t *socket, byte *ip, mp_uint_t port, int *_errno) {
    // use "bind" function to open the socket in client mode
    if (wiznet5k_socket_bind(socket, ip, 0, _errno) != 0) {
        return -1;
    }

    // now connect
    MP_THREAD_GIL_EXIT();
    mp_int_t ret = WIZCHIP_EXPORT(connect)(socket->fileno, ip, port);
    MP_THREAD_GIL_ENTER();

    if (ret < 0) {
        wiznet5k_socket_close(socket);
        *_errno = -ret;
        return -1;
    }

    // success
    return 0;
}

/*
 * CLAUDE FIX: Rewritten socket_send to be synchronous
 *
 * For large transfers (>= 1KB), this function now:
 * 1. Queues the transfer
 * 2. WAITS for the transfer to complete
 * 3. Returns actual bytes sent (or error)
 *
 * This ensures buffer safety (caller's buffer won't be freed while DMA is running)
 * and proper error propagation.
 */
static mp_uint_t wiznet5k_socket_send(mod_network_socket_obj_t *socket, const byte *buf, mp_uint_t len, int *_errno) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_send: socket=%d, len=%d\n", sn, len);

    // For small transfers (< 1KB), use original blocking method
    if (len < 1024) {
        MP_THREAD_GIL_EXIT();
        mp_int_t ret = WIZCHIP_EXPORT(send)(socket->fileno, (byte *)buf, len);
        MP_THREAD_GIL_ENTER();

        if (ret < 0) {
            DEBUG_PRINT("socket_send: small transfer failed, ret=%d\n", ret);
            wiznet5k_socket_close(socket);
            *_errno = -ret;
            return -1;
        }
        DEBUG_PRINT("socket_send: small transfer success, sent %d bytes\n", ret);
        return ret;
    }

    // For large transfers (>= 1KB), use DMA with circular buffer
    DEBUG_PRINT("socket_send: using DMA transfer for %d bytes\n", len);

    // Get transfer descriptor
    w5500_tx_transfer_t* transfer = w5500_get_transfer();
    if (!transfer) {
        DEBUG_PRINT("socket_send: transfer pool exhausted, falling back to blocking\n");
        MP_THREAD_GIL_EXIT();
        mp_int_t ret = WIZCHIP_EXPORT(send)(socket->fileno, (byte *)buf, len);
        MP_THREAD_GIL_ENTER();

        if (ret < 0) {
            wiznet5k_socket_close(socket);
            *_errno = -ret;
            return -1;
        }
        return ret;
    }

    // Initialize transfer descriptor
    transfer->data_ptr = buf;
    transfer->total_size = len;
    transfer->bytes_sent = 0;
    transfer->socket_num = sn;
    transfer->state = TX_WAITING_BUFFER_SPACE;

    // Initial chunk size based on available buffer space
    uint16_t free_space = getSn_TX_FSR(sn);
    uint16_t max_chunk = getSn_TxMAX(sn);
    transfer->chunk_size = (len < free_space) ? len : free_space;
    if (transfer->chunk_size > max_chunk) {
        transfer->chunk_size = max_chunk;
    }

    // Queue transfer
    w5500_queue_transfer(transfer);

    DEBUG_PRINT("socket_send: queued transfer, initial chunk size=%d\n", transfer->chunk_size);

    // CLAUDE FIX: Wait for transfer to complete (synchronous!)
    // Use socket timeout or a reasonable default
    uint32_t timeout_ms = (socket->timeout > 0) ? socket->timeout : 30000; // 30 second default

    int result = w5500_wait_transfer_complete(transfer, timeout_ms);

    if (result < 0 || transfer->state == TX_ERROR) {
        DEBUG_PRINT("socket_send: transfer failed\n");
        // Transfer already dequeued and released by w5500_tx_service
        *_errno = MP_EIO;
        return -1;
    }

    uint32_t bytes_sent = transfer->bytes_sent;
    // Transfer already dequeued and released by w5500_tx_service

    DEBUG_PRINT("socket_send: transfer complete, sent %d bytes\n", bytes_sent);
    return bytes_sent;
}

static mp_uint_t wiznet5k_socket_recv(mod_network_socket_obj_t *socket, byte *buf, mp_uint_t len, int *_errno) {
    uint8_t sn = (uint8_t)socket->fileno;
    DEBUG_PRINT("socket_recv: socket=%d, len=%d, timeout=%d\n",
                sn, len, socket->timeout);

    uint32_t start_time = mp_hal_ticks_ms();
    int loop_counter = 0;

    while (1) {
        // First, check if data is available
        uint16_t rx_rsr = getSn_RX_RSR(sn);
        if (loop_counter % 100 == 0) {
            DEBUG_PRINT("socket_recv: checking, RX_RSR=%d\n", rx_rsr);
        }

        if (rx_rsr > 0) {
            // Data is available, call recv
            DEBUG_PRINT("socket_recv: data available (%d bytes), calling recv\n", rx_rsr);

            MP_THREAD_GIL_EXIT();
            mp_int_t ret = WIZCHIP_EXPORT(recv)(sn, buf, len);
            MP_THREAD_GIL_ENTER();

            DEBUG_PRINT("socket_recv: WIZCHIP_EXPORT(recv) returned %d\n", ret);

            if (ret < 0) {
                DEBUG_PRINT("socket_recv: recv failed: %d\n", ret);
                wiznet5k_socket_close(socket);
                *_errno = -ret;
                return -1;
            }

            DEBUG_PRINT("socket_recv: success, received %d bytes\n", ret);
            return ret;
        }

        // Check if socket is still connected
        uint8_t sr = getSn_SR(sn);
        if (loop_counter % 100 == 0) {
            DEBUG_PRINT("socket_recv: socket state=0x%02x (%s)\n",
                       sr, socket_state_str(sr));
        }

        if (sr != SOCK_ESTABLISHED && sr != SOCK_CLOSE_WAIT) {
            DEBUG_PRINT("socket_recv: socket not in valid state for recv\n");
            *_errno = MP_ENOTCONN;
            return -1;
        }

        // Check for timeout
        if (socket->timeout > 0) {
            uint32_t elapsed = mp_hal_ticks_ms() - start_time;
            if (elapsed >= socket->timeout) {
                DEBUG_PRINT("socket_recv: timeout after %d ms\n", elapsed);
                *_errno = MP_ETIMEDOUT;
                return -1;
            }
        } else if (socket->timeout == 0) {
            // Non-blocking mode - return immediately if no data
            DEBUG_PRINT("socket_recv: non-blocking mode, no data available\n");
            *_errno = MP_EAGAIN;
            return -1;
        }

        // Small delay to prevent busy-waiting
        mp_hal_delay_ms(10);
        loop_counter++;
    }
}


static mp_uint_t wiznet5k_socket_sendto(mod_network_socket_obj_t *socket, const byte *buf, mp_uint_t len, byte *ip, mp_uint_t port, int *_errno) {
    if (socket->domain == 0) {
        // socket not opened; use "bind" function to open the socket in client mode
        if (wiznet5k_socket_bind(socket, ip, 0, _errno) != 0) {
            return -1;
        }
    }

    MP_THREAD_GIL_EXIT();
    mp_int_t ret = WIZCHIP_EXPORT(sendto)(socket->fileno, (byte *)buf, len, ip, port);
    MP_THREAD_GIL_ENTER();

    if (ret < 0) {
        wiznet5k_socket_close(socket);
        *_errno = -ret;
        return -1;
    }
    return ret;
}

    uint16_t port2;
    MP_THREAD_GIL_EXIT();
    mp_int_t ret = WIZCHIP_EXPORT(recvfrom)(socket->fileno, buf, len, ip, &port2);
    MP_THREAD_GIL_ENTER();
    *port = port2;
    if (ret < 0) {
        wiznet5k_socket_close(socket);
        *_errno = -ret;
        return -1;
    }
    return ret;
}

static int wiznet5k_socket_setsockopt(mod_network_socket_obj_t *socket, mp_uint_t level, mp_uint_t opt, const void *optval, mp_uint_t optlen, int *_errno) {
    DEBUG_PRINT("socket_setsockopt: socket=%d, level=%d, opt=%d, optlen=%d\n",
               socket->fileno, level, opt, optlen);

    if (level == 1) {  // SOL_SOCKET
        switch (opt) {
            case 2:  // SO_REUSEADDR
                DEBUG_PRINT("socket_setsockopt: SO_REUSEADDR - accepting as no-op\n");
                return 0;  // Accept as no-op

            case 20: // SO_RCVTIMEO or similar (timeout)
                DEBUG_PRINT("socket_setsockopt: timeout option\n");
                // Might be setting timeout via setsockopt instead of settimeout
                return 0;  // Accept

            default:
                DEBUG_PRINT("socket_setsockopt: unsupported SOL_SOCKET option %d\n", opt);
                *_errno = MP_EINVAL;
                return -1;
        }
    }

    DEBUG_PRINT("socket_setsockopt: unsupported level %d\n", level);
    *_errno = MP_EINVAL;
    return -1;
}

static int wiznet5k_socket_ioctl(mod_network_socket_obj_t *socket, mp_uint_t request, mp_uint_t arg, int *_errno) {
    static int ioctl_call_count = 0;
    ioctl_call_count++;
    if (ioctl_call_count % 100 == 0) {
        DEBUG_PRINT("socket_ioctl: request=0x%x, arg=0x%x, socket=%d\n",
                    request, arg, socket->fileno);
    }

    if (request == MP_STREAM_POLL) {
        int ret = 0;
        uint8_t sn = (uint8_t)socket->fileno;

        // Check RX buffer
        uint16_t rx_rsr = getSn_RX_RSR(sn);
        uint16_t tx_fsr = getSn_TX_FSR(sn);
        uint8_t sr = getSn_SR(sn);

        if (ioctl_call_count % 100 == 0) {
            DEBUG_PRINT("socket_ioctl: socket %d - state=0x%02x (%s), RX_RSR=%d, TX_FSR=%d\n",
                       sn, sr, socket_state_str(sr), rx_rsr, tx_fsr);
        }

        if (arg & MP_STREAM_POLL_RD) {
            if (sr == SOCK_LISTEN) {
                // For listening sockets, check interrupt register for pending connections
                uint8_t ir = getSn_IR(sn);
                if (ioctl_call_count % 100 == 0) {
                    DEBUG_PRINT("socket_ioctl: listening socket, ir=0x%02x\n", ir);
                }
                if (ir & Sn_IR_CON) {
                    ret |= MP_STREAM_POLL_RD;
                    if (ioctl_call_count % 100 == 0) {
                        DEBUG_PRINT("socket_ioctl: connection pending!\n");
                    }
                }
            } else if (sr == SOCK_ESTABLISHED || sr == SOCK_CLOSE_WAIT) {
                // Check for received data
                uint16_t rx_rsr = getSn_RX_RSR(sn);
                if (rx_rsr > 0) {
                    ret |= MP_STREAM_POLL_RD;
                    if (ioctl_call_count % 100 == 0) {
                        DEBUG_PRINT("socket_ioctl: data available (RX_RSR=%d)\n", rx_rsr);
                    }
                }
            }
        }

        if (arg & MP_STREAM_POLL_WR) {
            if (sr == SOCK_ESTABLISHED || sr == SOCK_CLOSE_WAIT) {
                uint16_t tx_fsr = getSn_TX_FSR(sn);
                if (tx_fsr > 0) {
                    ret |= MP_STREAM_POLL_WR;
                    if (ioctl_call_count % 100 == 0) {
                        DEBUG_PRINT("socket_ioctl: can write (TX_FSR=%d)\n", tx_fsr);
                    }
                }
            } else if (sr == SOCK_INIT || sr == SOCK_LISTEN) {
                // Socket can accept writes (for connecting/listening)
                ret |= MP_STREAM_POLL_WR;
                if (ioctl_call_count % 100 == 0) {
                    DEBUG_PRINT("socket_ioctl: socket ready for connect/listen\n");
                }
            }
        }

        if (ioctl_call_count % 100 == 0) {
            DEBUG_PRINT("socket_ioctl: returning 0x%x\n", ret);
        }
        return ret;
    } else {
        if (ioctl_call_count % 100 == 0) {
            DEBUG_PRINT("socket_ioctl: unsupported request 0x%x\n", request);
        }
        *_errno = MP_EINVAL;
        return MP_STREAM_ERROR;
    }
}

static void wiznet5k_dhcp_init(wiznet5k_obj_t *self) {
    uint8_t test_buf[2048];
    uint8_t ret = 0;
    uint8_t dhcp_retry = 0;

    while (ret != DHCP_IP_LEASED) {
        mp_uint_t timeout = mp_hal_ticks_ms() + 3000;
        DHCP_init(1, test_buf);

        while (1) {
            ret = DHCP_run();
            if (ret == DHCP_IP_LEASED) {
                break;
            } else if (ret == DHCP_FAILED || mp_hal_ticks_ms() > timeout) {
                dhcp_retry++;
                break;
            }
            mpy_wiznet_yield();
        }

        if (dhcp_retry > 3) {
            mp_raise_msg(&mp_type_OSError, MP_ERROR_TEXT("timeout waiting for DHCP to get IP address"));
            break;
        }
    }

    if (ret == DHCP_IP_LEASED) {
        ctlnetwork(CN_GET_NETINFO, &self->netinfo);
    }
}

// WIZNET5K(spi, pin_cs, pin_rst[, pin_intn])
// Create and return a WIZNET5K object.
static mp_obj_t wiznet5k_make_new(const mp_obj_type_t *type, size_t n_args, size_t n_kw, const mp_obj_t *args) {
    mp_obj_base_t *spi;
    mp_hal_pin_obj_t cs;
    mp_hal_pin_obj_t rst;

    #if WIZNET5K_WITH_LWIP_STACK
    mp_hal_pin_obj_t pin_intn = (mp_hal_pin_obj_t)NULL;
    bool use_interrupt = false;

    // Bring down interface while configuring
    wiznet5k_obj.netif.flags = 0;
    #endif

    #ifdef MICROPY_HW_WIZNET_SPI_ID
    // Allow auto-configuration of SPI if defined for board and no args passed
    if (n_args == 0 && n_kw == 0) {
        // Initialize SPI.
        mp_obj_t spi_obj = MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_SPI_SCK);
        mp_obj_t miso_obj = MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_SPI_MISO);
        mp_obj_t mosi_obj = MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_SPI_MOSI);
        mp_obj_t args[] = {
            MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_SPI_ID),
            MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_SPI_BAUDRATE),
            MP_ROM_QSTR(MP_QSTR_sck), mp_pin_make_new(NULL, 1, 0, &spi_obj),
            MP_ROM_QSTR(MP_QSTR_miso), mp_pin_make_new(NULL, 1, 0, &miso_obj),
            MP_ROM_QSTR(MP_QSTR_mosi), mp_pin_make_new(NULL, 1, 0, &mosi_obj),
        };
        spi = MP_OBJ_TO_PTR(MP_OBJ_TYPE_GET_SLOT(&machine_spi_type, make_new)((mp_obj_t)&machine_spi_type, 2, 3, args));

        cs = mp_hal_get_pin_obj(mp_pin_make_new(NULL, 1, 0, (mp_obj_t[]) {MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_PIN_CS)}));
        rst = mp_hal_get_pin_obj(mp_pin_make_new(NULL, 1, 0, (mp_obj_t[]) {MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_PIN_RST)}));
        #if WIZNET5K_WITH_LWIP_STACK && defined(MICROPY_HW_WIZNET_PIN_INTN)
        pin_intn = mp_hal_get_pin_obj(mp_pin_make_new(NULL, 1, 0, (mp_obj_t[]) {MP_OBJ_NEW_SMALL_INT(MICROPY_HW_WIZNET_PIN_INTN)}));
        use_interrupt = true;
        #endif

    } else
    #endif
    {
        // If passing in args, must supply spi, pin_cs, pin_rst and optionally pin_intn
        #if WIZNET5K_WITH_LWIP_STACK
        mp_arg_check_num(n_args, n_kw, 3, 4, false);
        #else
        mp_arg_check_num(n_args, n_kw, 3, 3, false);
        #endif
        spi = mp_hal_get_spi_obj(args[0]);
        cs = mp_hal_get_pin_obj(args[1]);
        rst = mp_hal_get_pin_obj(args[2]);
        #if WIZNET5K_WITH_LWIP_STACK
        if (n_args > 3) {
            pin_intn = mp_hal_get_pin_obj(args[3]);
            use_interrupt = true;
        }
        #endif
    }

    mp_hal_pin_output(cs);
    mp_hal_pin_output(rst);

    // Init the wiznet5k object
    wiznet5k_obj.base.type = (mp_obj_type_t *)&mod_network_nic_type_wiznet5k;
    wiznet5k_obj.cris_state = 0;
    wiznet5k_obj.spi = spi;
    wiznet5k_obj.spi_transfer = ((mp_machine_spi_p_t *)MP_OBJ_TYPE_GET_SLOT(spi->type, protocol))->transfer;
    wiznet5k_obj.cs = cs;
    wiznet5k_obj.rst = rst;
    #if WIZNET5K_WITH_LWIP_STACK
    wiznet5k_obj.pin_intn = pin_intn;
    wiznet5k_obj.use_interrupt = use_interrupt;
    wiznet5k_obj.trace_flags = 0;
    #else // WIZNET5K_PROVIDED_STACK
    wiznet5k_obj.active = false;
    wiznet5k_obj.socket_used = 0;
    #endif

    // Return wiznet5k object
    return MP_OBJ_FROM_PTR(&wiznet5k_obj);
}

static mp_obj_t wiznet5k_active(size_t n_args, const mp_obj_t *args) {
    wiznet5k_obj_t *self = MP_OBJ_TO_PTR(args[0]);
    DEBUG_PRINT("wiznet5k_active: n_args=%d, IS_ACTIVE=%d\n",
               n_args, IS_ACTIVE(self));

    if (n_args == 1) {
        DEBUG_PRINT("wiznet5k_active: getting state\n");
        return mp_obj_new_bool(IS_ACTIVE(self));
    } else {
        if (mp_obj_is_true(args[1])) {
            if (!IS_ACTIVE(self)) {
                DEBUG_PRINT("wiznet5k_active: activating...\n");

                /*!< Wiznet initialisation */
                // Reset the chip
                mp_hal_pin_low(wiznet5k_obj.rst);
                mp_hal_delay_ms(1); // datasheet says 2us
                mp_hal_pin_high(wiznet5k_obj.rst);
                mp_hal_delay_ms(160); // datasheet says 150ms

                // CLAUDE FIX: Claim DMA channel BEFORE registering SPI callbacks
                // This ensures dma_tx_chan is valid if wiz_spi_writeburst is called
                w5500_dma_init();

                // Set physical interface callbacks
                reg_wizchip_cris_cbfunc(wiz_cris_enter, wiz_cris_exit);
                reg_wizchip_cs_cbfunc(wiz_cs_select, wiz_cs_deselect);
                reg_wizchip_spi_cbfunc(wiz_spi_readbyte, wiz_spi_writebyte);
                reg_wizchip_spiburst_cbfunc(wiz_spi_read, wiz_spi_writeburst);

                // SPI DMA will be enabled only during DMA transfers

                // Configure lwip/provided specific settings
                DEBUG_PRINT("wiznet5k_active: calling wiznet5k_init()\n");
                wiznet5k_init();

                // If the device doesn't have a MAC address then set one
                #if WIZNET5K_WITH_LWIP_STACK
                uint8_t *mac = self->netif.hwaddr;
                #else // WIZNET5K_PROVIDED_STACK
                uint8_t *mac = wiznet5k_obj.netinfo.mac;
                #endif
                getSHAR(mac);
                if ((mac[0] | mac[1] | mac[2] | mac[3] | mac[4] | mac[5]) == 0) {
                    mp_hal_get_mac(MP_HAL_MAC_ETH0, mac);
                    setSHAR(mac);
                }

                #if WIZNET5K_WITH_LWIP_STACK && LWIP_IPV6
                netif_create_ip6_linklocal_address(&self->netif, 1);
                #endif

                // seems we need a small delay after init
                mp_hal_delay_ms(250);

            }
        } else {
            DEBUG_PRINT("wiznet5k_active: deactivating...\n");
            #if WIZNET5K_WITH_LWIP_STACK
            netif_set_down(&self->netif);
            netif_set_link_down(&self->netif);
            wiznet5k_deinit();
            #else // WIZNET5K_PROVIDED_STACK
            self->active = false;
            wizchip_sw_reset();
            #endif
        }
        return mp_const_none;
    }
}
static MP_DEFINE_CONST_FUN_OBJ_VAR_BETWEEN(wiznet5k_active_obj, 1, 2, wiznet5k_active);

#if WIZNET5K_PROVIDED_STACK
// ifconfig([(ip, subnet, gateway, dns)])
// Get/set IP address, subnet mask, gateway and DNS.
static mp_obj_t wiznet5k_ifconfig(size_t n_args, const mp_obj_t *args) {
    wiz_NetInfo netinfo;
    wiznet5k_obj_t *self = MP_OBJ_TO_PTR(args[0]);
    ctlnetwork(CN_GET_NETINFO, &netinfo);
    if (n_args == 1) {
        // Get IP addresses
        mp_obj_t tuple[4] = {
            netutils_format_ipv4_addr(netinfo.ip, NETUTILS_BIG),
            netutils_format_ipv4_addr(netinfo.sn, NETUTILS_BIG),
            netutils_format_ipv4_addr(netinfo.gw, NETUTILS_BIG),
            netutils_format_ipv4_addr(netinfo.dns, NETUTILS_BIG),
        };
        return mp_obj_new_tuple(4, tuple);
    } else if (args[1] == MP_OBJ_NEW_QSTR(MP_QSTR_dhcp)) {
        // Start the DHCP client
        self->netinfo.dhcp = NETINFO_DHCP;
        wiznet5k_dhcp_init((void *)self);
        mp_obj_t tuple[4] = {
            netutils_format_ipv4_addr(self->netinfo.ip, NETUTILS_BIG),
            netutils_format_ipv4_addr(self->netinfo.sn, NETUTILS_BIG),
            netutils_format_ipv4_addr(self->netinfo.gw, NETUTILS_BIG),
            netutils_format_ipv4_addr(self->netinfo.dns, NETUTILS_BIG),

        };
        return mp_obj_new_tuple(4, tuple);

    } else {
        // Set static IP addresses
        self->netinfo.dhcp = NETINFO_STATIC;
        mp_obj_t *items;
        mp_obj_get_array_fixed_n(args[1], 4, &items);
        netutils_parse_ipv4_addr(items[0], netinfo.ip, NETUTILS_BIG);
        netutils_parse_ipv4_addr(items[1], netinfo.sn, NETUTILS_BIG);
        netutils_parse_ipv4_addr(items[2], netinfo.gw, NETUTILS_BIG);
        netutils_parse_ipv4_addr(items[3], netinfo.dns, NETUTILS_BIG);
        ctlnetwork(CN_SET_NETINFO, &netinfo);
        return mp_const_none;
    }
}
static MP_DEFINE_CONST_FUN_OBJ_VAR_BETWEEN(wiznet5k_ifconfig_obj, 1, 2, wiznet5k_ifconfig);

static mp_obj_t wiznet5k_status(size_t n_args, const mp_obj_t *args) {
    wiznet5k_obj_t *self = MP_OBJ_TO_PTR(args[0]);
    (void)self;

            } else {

}

static mp_obj_t wiznet5k_config(size_t n_args, const mp_obj_t *args, mp_map_t *kwargs) {
    wiznet5k_obj_t *self = MP_OBJ_TO_PTR(args[0]);

    if (kwargs->used == 0) {
        // Get config value
        if (n_args != 2) {
            mp_raise_TypeError(MP_ERROR_TEXT("must query one param"));
        }

        switch (mp_obj_str_get_qstr(args[1])) {
            case MP_QSTR_mac: {
                uint8_t buf[6];
                wiznet5k_get_mac_address(self, buf);
                return mp_obj_new_bytes(buf, 6);
            }
            default:
                mp_raise_ValueError(MP_ERROR_TEXT("unknown config param"));
        }
    } else {
        // Set config value(s)
        if (n_args != 1) {
            mp_raise_TypeError(MP_ERROR_TEXT("can't specify pos and kw args"));
        }

        for (size_t i = 0; i < kwargs->alloc; ++i) {
            if (MP_MAP_SLOT_IS_FILLED(kwargs, i)) {
                mp_map_elem_t *e = &kwargs->table[i];
                switch (mp_obj_str_get_qstr(e->key)) {
                    case MP_QSTR_mac: {
                        mp_buffer_info_t buf;
                        mp_get_buffer_raise(e->value, &buf, MP_BUFFER_READ);
                        if (buf.len != 6) {
                            mp_raise_ValueError(NULL);
                        }
                        setSHAR(buf.buf);
                        #if WIZNET5K_WITH_LWIP_STACK
                        memcpy(self->netif.hwaddr, buf.buf, 6);
                        #endif
                        break;
                    }
                    #if WIZNET5K_WITH_LWIP_STACK
                    case MP_QSTR_trace: {
                        self->trace_flags = mp_obj_get_int(e->value);
                        break;
                    }
                    #endif
                    default:
                        mp_raise_ValueError(MP_ERROR_TEXT("unknown config param"));
                }
            }
        }

        return mp_const_none;
    }
}
static MP_DEFINE_CONST_FUN_OBJ_KW(wiznet5k_config_obj, 1, wiznet5k_config);
